Returned, not printed, so the plain-text caller does no HTML escaping. */ -}} {{- $url := .Permalink -}} {{- with .OutputFormats.Get "markdown" }}{{ $url = .Permalink }}{{ end -}} {{- $entry := printf "- [%s](%s)" .Title $url -}} {{- with .Description }}{{ $entry = printf "%s: %s" $entry . }}{{ end -}} {{- return $entry -}}
